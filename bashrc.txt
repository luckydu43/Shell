# Les tokens sont stockés dans ~/.tokens.env
#TOKEN_ACCESS_CLIENT_BACKEND_WEB=...

token() {
  local tokens_env_file="$HOME/.tokens.env"
  if [ ! -f "$tokens_env_file" ]; then
    echo "Fichier de tokens introuvable : $tokens_env_file"
    return 1
  fi
  set -a
  . "$tokens_env_file"
  set +a
  ./ci/migrate.sh
}

ajout_var_tomcat() {
  local var_name="$1"
  local value="$2"
  local current="${!var_name:-}"
  case ":$current:" in
    *":$value:"*) ;;
    *) export "$var_name=${current:+$current:}$value" ;;
  esac
}

# Git / fichiers
# Pour remettre un git checkout -B qui a reset
# git reflog show fix/CLIENTTMASIG-2886
# git reset --hard fix/CLIENTTMASIG-2886@{1}
archive() {
  git fsck --lost-found \
    | grep 'dangling blob' \
    | awk '{print $3}' \
    | while read -r sha; do
        git show "$sha" | batcat -p
      done
}

}

# NAS
nas() {
  if ! dpkg -s nfs-common >/dev/null 2>&1; then
    sudo apt update
    sudo apt install -y nfs-common
  fi
  sudo mkdir -p /NAS
  if ! grep -q '^vm1\.domaineEntreprise\.fr:/NAS[[:space:]]\+/NAS[[:space:]]\+nfs' /etc/fstab; then
    echo 'vm1.domaineEntreprise.fr:/NAS /NAS nfs defaults,_netdev 0 0' | sudo tee -a /etc/fstab
  fi
  sudo mount -a
}

# Hosts
ajout_hote() {
  local ip="$1"
  local host="$2"
  if grep -Eq "^[[:space:]]*$ip[[:space:]]+$host([[:space:]]|$)" /etc/hosts; then
    return 0
  fi
  echo "$ip $host" | sudo tee -a /etc/hosts >/dev/null
  echo "Ajouté : $ip $host"
}
